

******CREATION DES TABLES******



CREATE TABLE FACTURES_FNE (
    ID INT IDENTITY PRIMARY KEY,
    DO_PIECE VARCHAR(50),
    FNE_INVOICE_ID VARCHAR(255),
    FNE_REFERENCE VARCHAR(255),
    DATE_CERTIFICATION DATETIME,
    STATUT VARCHAR(50)
    MESSAGE_API VARCHAR(MAX)
)


CREATE TABLE FACTURES_FNE_ITEMS (
    ID INT IDENTITY PRIMARY KEY,
    DO_PIECE VARCHAR(50),
    ARTICLE_REF VARCHAR(100),
    FNE_ITEM_ID VARCHAR(255),
    QUANTITE FLOAT
)




CREATE TABLE AVOIRS (
    ID INT IDENTITY PRIMARY KEY,
    FNE_INVOICE_ID VARCHAR(100) NOT NULL,
    DO_PIECE VARCHAR(50),
    DATE_AVOIR DATETIME DEFAULT GETDATE(),
    TOTAL_QTY DECIMAL(18,2) DEFAULT 0,
    STATUS VARCHAR(20) DEFAULT 'CERTIFIE'
);



CREATE TABLE AVOIR_ITEMS (
    ID INT IDENTITY PRIMARY KEY,
    AVOIR_ID INT,
    ITEM_ID VARCHAR(100),
    QUANTITE DECIMAL(18,2)
);


CREATE TABLE ACHATS_FNE (

    ID INT IDENTITY(1,1) PRIMARY KEY,

    DO_PIECE VARCHAR(50),

    FNE_INVOICE_ID VARCHAR(255),

    FNE_REFERENCE VARCHAR(255),

    FOURNISSEUR VARCHAR(255),

    MONTANT_TTC DECIMAL(18,2),

    DATE_CERTIFICATION DATETIME DEFAULT GETDATE()
)
